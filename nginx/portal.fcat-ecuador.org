# Detect social media / messaging crawlers for link previews
map $http_user_agent $is_social_crawler {
    default                         0;
    "~*facebookexternalhit"         1;
    "~*Facebot"                     1;
    "~*Twitterbot"                  1;
    "~*LinkedInBot"                 1;
    "~*WhatsApp"                    1;
    "~*TelegramBot"                 1;
    "~*Slackbot"                    1;
    "~*Discordbot"                  1;
    "~*Applebot"                    1;
    "~*iMessageBot"                 1;
}

# Rate limiting for unauthenticated public endpoints
limit_req_zone $binary_remote_addr zone=public_api:10m rate=10r/s;
limit_req_zone $binary_remote_addr zone=public_pages:10m rate=5r/s;

# HTTP to HTTPS redirect
server {
    listen 80;
    server_name portal.fcat-ecuador.org;
    return 301 https://$server_name$request_uri;
}

# HTTPS server block with Google OAuth via oauth2-proxy
server {
    listen 443 ssl http2;
    server_name portal.fcat-ecuador.org;

    # SSL Configuration (managed by certbot)
    ssl_certificate /etc/letsencrypt/live/portal.fcat-ecuador.org/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/portal.fcat-ecuador.org/privkey.pem;

    # SSL Security Headers
    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_ciphers ECDHE-RSA-AES128-GCM-SHA256:ECDHE-RSA-AES256-GCM-SHA384;
    ssl_prefer_server_ciphers off;
    add_header Strict-Transport-Security "max-age=63072000" always;

    client_max_body_size 50m;

    # ===== oauth2-proxy endpoints =====
    location /oauth2/ {
        proxy_pass http://127.0.0.1:4181;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    location = /oauth2/auth {
        proxy_pass http://127.0.0.1:4181;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_pass_request_body off;
        proxy_set_header Content-Length "";
    }

    # ===== Public assets (no auth) — used for OG image in link previews =====
    location = /logo-fcat.png {
        proxy_pass http://127.0.0.1:3003;
        proxy_set_header Host $host;
        expires 7d;
    }

    # Site icons. Next's app-router emits <link href="/favicon.ico?...">, and the
    # browser requests them on the logged-out public pages too. Under `location /`
    # they 401 → oauth sign-in HTML served as an .ico, so the tab icon breaks.
    # (nginx `location =` ignores the query string, so the hashed URLs match.)
    location = /favicon.ico {
        proxy_pass http://127.0.0.1:3003;
        proxy_set_header Host $host;
        expires 7d;
    }

    location = /icon.svg {
        proxy_pass http://127.0.0.1:3003;
        proxy_set_header Host $host;
        expires 7d;
    }

    # ===== Public pages (no auth) — share links, public dashboard =====
    location /public/ {
        limit_req zone=public_pages burst=15 nodelay;
        proxy_pass http://127.0.0.1:3003;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header X-Forwarded-Host $host;
        # NO auth_request — intentionally public
        # NO X-Forwarded-Email — anonymous access
    }

    # ===== Public API (no auth) — serves images for share links =====
    location /api/public/ {
        limit_req zone=public_api burst=30 nodelay;
        proxy_pass http://127.0.0.1:3003;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header X-Forwarded-Host $host;
        # NO auth_request — token-validated at application level
    }

    # ===== Public page static assets (no auth) =====
    # The public overview page (/public/biochoco-overview) is served from the
    # exempt /public/ location, but the browser then loads its client JS from
    # /_next/static/* and its media (hero/habitat/gallery images, reserve.geojson)
    # from /biochoco-overview/*. Both must be reachable WITHOUT oauth or a
    # logged-out visitor gets those 401'd → redirected to Google sign-in → the
    # page never hydrates (blank map, dead spectrograms) and images break.
    # Compiled Next.js bundles are public build artifacts (no secrets; server
    # actions keep their own auth), so exposing them is safe and standard.
    location /_next/static/ {
        proxy_pass http://127.0.0.1:3003;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header X-Forwarded-Host $host;
        # NO auth_request — public build artifacts; no rate limit (a page pulls many chunks)
    }

    location /_next/image {
        proxy_pass http://127.0.0.1:3003;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header X-Forwarded-Host $host;
        # NO auth_request — Next image optimizer for public assets
    }

    location /biochoco-overview/ {
        limit_req zone=public_pages burst=30 nodelay;
        proxy_pass http://127.0.0.1:3003;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header X-Forwarded-Host $host;
        # NO auth_request — intentionally-public marketing media
    }

    # ===== Field Uploader API (no oauth2 — Bearer-token validated in the app) =====
    # The desktop field-uploader is a machine client; it can't do an interactive
    # Google login. It authenticates with FIELD_UPLOAD_TOKEN (Bearer), checked in
    # src/lib/field-upload-auth.ts (timing-safe) on every request, plus a per-IP
    # app-level rate limit. So this path must bypass oauth2-proxy, exactly like
    # /api/public/ above. Keeps the Authorization header (no auth_request to strip
    # it); does NOT set X-Forwarded-Email (no user identity here).
    location /api/field-upload/ {
        limit_req zone=public_api burst=30 nodelay;
        proxy_pass http://127.0.0.1:3003;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header X-Forwarded-Host $host;
        # NO auth_request — token-validated at application level
    }

    # ===== Main app - requires authentication =====
    location / {
        # Serve OG metadata to social crawlers without requiring auth
        if ($is_social_crawler) {
            add_header Content-Type text/html;
            return 200 '<!DOCTYPE html><html lang="es"><head><meta charset="UTF-8"><title>Portal FCAT</title><meta property="og:title" content="Portal FCAT"><meta property="og:description" content="Plataforma interna de FCAT para personal y colaboradores"><meta property="og:image" content="https://portal.fcat-ecuador.org/logo-fcat.png"><meta property="og:image:width" content="1577"><meta property="og:image:height" content="889"><meta property="og:url" content="https://portal.fcat-ecuador.org"><meta property="og:type" content="website"><meta property="og:site_name" content="Portal FCAT"></head><body></body></html>';
        }

        # Check with oauth2-proxy before allowing access
        auth_request /oauth2/auth;
        error_page 401 = /oauth2/sign_in;

        # Pass user email to app (read by src/proxy.ts via X-Forwarded-Email)
        auth_request_set $email $upstream_http_x_auth_request_email;
        proxy_set_header X-Forwarded-Email $email;

        # Proxy to Next.js portal (port 3003 on host)
        proxy_pass http://127.0.0.1:3003;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header X-Forwarded-Host $host;

        # WebSocket support (Next.js HMR in dev, potential future use)
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";

        proxy_connect_timeout 60s;
        proxy_send_timeout 60s;
        proxy_read_timeout 300s;
    }
}
